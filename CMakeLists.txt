cmake_minimum_required(VERSION 3.28...3.30)
option(BROADCAST_TIMER_SYSTEM_BUILD "Build with installed OBS and Qt (Linux development)" OFF)
option(BROADCAST_TIMER_UI_ONLY "Build GUI tests without OBS" OFF)
if(NOT BROADCAST_TIMER_SYSTEM_BUILD AND NOT BROADCAST_TIMER_UI_ONLY)
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
endif()
project(masqrayd-broadcast-timer VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT BROADCAST_TIMER_SYSTEM_BUILD AND NOT BROADCAST_TIMER_UI_ONLY)
  set(ENABLE_FRONTEND_API ON CACHE BOOL "OBS frontend required" FORCE)
  set(ENABLE_QT ON CACHE BOOL "Qt required" FORCE)
  include(compilerconfig)
  include(defaults)
  include(helpers)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
if(NOT BROADCAST_TIMER_UI_ONLY)
  find_package(libobs REQUIRED)
  find_package(obs-frontend-api REQUIRED)
  add_library(masqrayd-broadcast-timer MODULE src/plugin-main.cpp src/timer-window.cpp src/broadcast-stack.cpp src/native-camera-surface.cpp)
  target_link_libraries(masqrayd-broadcast-timer PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets)
  if(BROADCAST_TIMER_SYSTEM_BUILD)
    set_target_properties(masqrayd-broadcast-timer PROPERTIES PREFIX "")
  else()
    set_target_properties_plugin(masqrayd-broadcast-timer PROPERTIES OUTPUT_NAME "masqrayd-broadcast-timer")
  endif()
endif()

# Optional coordinated Windows package. Partner source is checked out at pinned
# commits by CI; each plugin still owns its own windows and configuration.
if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/_partners/camera/Camera-Pop-Out-Native/src/plugin-main.cpp")
  add_library(camera-popout-native MODULE _partners/camera/Camera-Pop-Out-Native/src/plugin-main.cpp)
  target_compile_definitions(camera-popout-native PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_compile_options(camera-popout-native PRIVATE /utf-8)
  target_link_libraries(camera-popout-native PRIVATE OBS::libobs OBS::obs-frontend-api user32 gdi32 dwmapi)
  set_target_properties(camera-popout-native PROPERTIES PREFIX "" OUTPUT_NAME "camera-popout-native")
  install(TARGETS camera-popout-native RUNTIME DESTINATION "camera-popout-native/bin/64bit" LIBRARY DESTINATION "camera-popout-native/bin/64bit")

  add_library(scene-vault MODULE _partners/controller/src/plugin-main.cpp _partners/controller/src/scene-vault.cpp _partners/controller/src/scene-controller.cpp)
  target_link_libraries(scene-vault PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets)
  target_compile_options(scene-vault PRIVATE /utf-8)
  set_target_properties(scene-vault PROPERTIES PREFIX "" OUTPUT_NAME "scene-vault")
  install(TARGETS scene-vault RUNTIME DESTINATION "scene-vault/bin/64bit" LIBRARY DESTINATION "scene-vault/bin/64bit")
endif()

  target_include_directories(timer-state-tests PRIVATE src)
  add_test(NAME timer-state COMMAND timer-state-tests)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(timer-ui-tests tests/timer-ui-tests.cpp src/timer-window.cpp)
  target_include_directories(timer-ui-tests PRIVATE src)
  target_link_libraries(timer-ui-tests PRIVATE Qt6::Core Qt6::Widgets Qt6::Test)
  add_test(NAME timer-ui COMMAND timer-ui-tests)
  if(NOT WIN32)
    set_tests_properties(timer-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  add_executable(broadcast-stack-tests tests/broadcast-stack-tests.cpp src/broadcast-stack.cpp src/native-camera-surface.cpp src/timer-window.cpp)
  target_include_directories(broadcast-stack-tests PRIVATE src)
  target_link_libraries(broadcast-stack-tests PRIVATE Qt6::Core Qt6::Widgets Qt6::Test)
  if(WIN32)
    target_link_libraries(broadcast-stack-tests PRIVATE user32)
  endif()
  add_test(NAME broadcast-stack COMMAND broadcast-stack-tests)
  if(NOT WIN32)
    set_tests_properties(broadcast-stack PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
