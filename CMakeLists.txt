cmake_minimum_required(VERSION 3.28...3.30)
option(BROADCAST_TIMER_SYSTEM_BUILD "Build with installed OBS and Qt (Linux development)" OFF)
option(BROADCAST_TIMER_UI_ONLY "Build GUI tests without OBS" OFF)
if(NOT BROADCAST_TIMER_SYSTEM_BUILD AND NOT BROADCAST_TIMER_UI_ONLY)
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
endif()
project(masqrayd-broadcast-timer VERSION 1.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT BROADCAST_TIMER_SYSTEM_BUILD AND NOT BROADCAST_TIMER_UI_ONLY)
  set(ENABLE_FRONTEND_API ON CACHE BOOL "OBS frontend required" FORCE)
  set(ENABLE_QT ON CACHE BOOL "Qt required" FORCE)
  include(compilerconfig)
  include(defaults)
  include(helpers)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
if(NOT BROADCAST_TIMER_UI_ONLY)
  find_package(libobs REQUIRED)
  find_package(obs-frontend-api REQUIRED)
  add_library(masqrayd-broadcast-timer MODULE src/plugin-main.cpp src/timer-window.cpp)
  target_link_libraries(masqrayd-broadcast-timer PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Core Qt6::Widgets)
  if(BROADCAST_TIMER_SYSTEM_BUILD)
    set_target_properties(masqrayd-broadcast-timer PROPERTIES PREFIX "")
  else()
    set_target_properties_plugin(masqrayd-broadcast-timer PROPERTIES OUTPUT_NAME "masqrayd-broadcast-timer")
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(timer-state-tests tests/timer-state-tests.cpp)
  target_include_directories(timer-state-tests PRIVATE src)
  add_test(NAME timer-state COMMAND timer-state-tests)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(timer-ui-tests tests/timer-ui-tests.cpp src/timer-window.cpp)
  target_include_directories(timer-ui-tests PRIVATE src)
  target_link_libraries(timer-ui-tests PRIVATE Qt6::Core Qt6::Widgets Qt6::Test)
  add_test(NAME timer-ui COMMAND timer-ui-tests)
  set_tests_properties(timer-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
